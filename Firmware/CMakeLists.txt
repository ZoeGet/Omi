cmake_minimum_required(VERSION 3.16.0)

set(PROJECT_VER "0.1.0")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(omipet)

#  仅隔离实验构建替换一份 SDK 源文件，默认不启用 / Replace one SDK source copy only in isolated experiments; disabled by default
option(OMIPET_USB_SOF_FIX "Enable the audited experimental USB SOF counter fix" OFF)
if(OMIPET_USB_SOF_FIX)
    idf_component_get_property(omipet_arduino_lib framework-arduinoespressif32 COMPONENT_LIB)
    get_target_property(omipet_arduino_dir ${omipet_arduino_lib} SOURCE_DIR)
    get_target_property(omipet_arduino_sources ${omipet_arduino_lib} SOURCES)
    set(omipet_sof_replacements 0)
    set(omipet_sof_sources "")
    foreach(omipet_source IN LISTS omipet_arduino_sources)
        if(omipet_source MATCHES "(^|/)cores/esp32/HWCDC\\.cpp$")
            get_filename_component(omipet_hwcdc "${omipet_source}" ABSOLUTE BASE_DIR "${omipet_arduino_dir}")
            set(omipet_sof_copy "${CMAKE_BINARY_DIR}/omipet-sof/HWCDC.cpp")
            execute_process(COMMAND "${PYTHON}" "${CMAKE_CURRENT_LIST_DIR}/scripts/usb_sof_fix.py"
                --source "${omipet_hwcdc}" --output "${omipet_sof_copy}"
                RESULT_VARIABLE omipet_sof_result)
            if(NOT omipet_sof_result EQUAL 0)
                message(FATAL_ERROR "Audited USB SOF source generation failed")
            endif()
            set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
                "${omipet_hwcdc}" "${CMAKE_CURRENT_LIST_DIR}/scripts/usb_sof_fix.py")
            list(APPEND omipet_sof_sources "${omipet_sof_copy}")
            math(EXPR omipet_sof_replacements "${omipet_sof_replacements}+1")
        else()
            list(APPEND omipet_sof_sources "${omipet_source}")
        endif()
    endforeach()
    if(NOT omipet_sof_replacements EQUAL 1)
        message(FATAL_ERROR "Expected exactly one HWCDC source in the Arduino component")
    endif()
    set_property(TARGET ${omipet_arduino_lib} PROPERTY SOURCES "${omipet_sof_sources}")
endif()
