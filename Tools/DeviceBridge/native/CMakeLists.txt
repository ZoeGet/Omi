cmake_minimum_required(VERSION 3.16)
project(omipet_pairing LANGUAGES C CXX)
if(NOT EXISTS "${MBEDTLS_SOURCE}/library/ecjpake.c")
  message(FATAL_ERROR "Supply the audited Mbed TLS 2.28.7 source directory")
endif()
file(STRINGS "${MBEDTLS_SOURCE}/include/mbedtls/version.h" version REGEX "^#define MBEDTLS_VERSION_STRING +\"2.28.7\"$")
if(NOT version)
  message(FATAL_ERROR "Expected Mbed TLS 2.28.7")
endif()
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(MBEDTLS_USER_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/config.h" CACHE STRING "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${MBEDTLS_SOURCE}" mbedtls)
add_library(omi_pake SHARED pake.cpp)
target_include_directories(omi_pake PRIVATE ../../../Firmware/include)
target_link_libraries(omi_pake PRIVATE mbedcrypto)
target_compile_features(omi_pake PRIVATE cxx_std_11)
